cmake_minimum_required(VERSION 3.20)
project(apsp_demo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(apsp INTERFACE)
target_include_directories(apsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(apsp INTERFACE /W4 /permissive-)
else()
  target_compile_options(apsp INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(apsp_demo src/main.cpp)
target_link_libraries(apsp_demo PRIVATE apsp)
set_target_properties(apsp_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
  add_executable(prueba_algoritmo test/prueba_algoritmo.cpp)
  target_link_libraries(prueba_algoritmo PRIVATE apsp)
  target_compile_definitions(prueba_algoritmo PRIVATE DATASET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/dataset")
  add_test(NAME algoritmo_y_carga COMMAND prueba_algoritmo)
endif()
